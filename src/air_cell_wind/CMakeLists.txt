cmake_minimum_required(VERSION 3.16)
project(air_cell_wind LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
set(GZ_SIM_MAJOR "8" CACHE STRING "Gazebo Sim major; must match the running server")
if(NOT GZ_SIM_MAJOR MATCHES "^(7|8|9)$")
  message(FATAL_ERROR "This source targets Gazebo Sim 7, 8 or 9. Fortress requires a separate port.")
endif()
find_package(gz-sim${GZ_SIM_MAJOR} REQUIRED)
if(GZ_SIM_MAJOR STREQUAL "7")
  set(TRANSPORT_VER 12)
  set(MSGS_VER 9)
  set(PLUGIN_VER 2)
elseif(GZ_SIM_MAJOR STREQUAL "8")
  set(TRANSPORT_VER 13)
  set(MSGS_VER 10)
  set(PLUGIN_VER 2)
else()
  set(TRANSPORT_VER 14)
  set(MSGS_VER 11)
  set(PLUGIN_VER 3)
endif()
find_package(gz-plugin${PLUGIN_VER} REQUIRED COMPONENTS register)
find_package(gz-transport${TRANSPORT_VER} REQUIRED)
find_package(gz-msgs${MSGS_VER} REQUIRED)
add_library(AirCellWind SHARED src/AirCellWind.cc)
target_compile_features(AirCellWind PRIVATE cxx_std_17)
target_compile_options(AirCellWind PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(AirCellWind PRIVATE
  gz-sim${GZ_SIM_MAJOR}::gz-sim${GZ_SIM_MAJOR}
  gz-plugin${PLUGIN_VER}::gz-plugin${PLUGIN_VER}
  gz-transport${TRANSPORT_VER}::gz-transport${TRANSPORT_VER}
  gz-msgs${MSGS_VER}::gz-msgs${MSGS_VER})
file(WRITE "${CMAKE_BINARY_DIR}/gazebo_major.txt" "${GZ_SIM_MAJOR}\n")

install(TARGETS AirCellWind LIBRARY DESTINATION lib)
install(FILES "${CMAKE_BINARY_DIR}/gazebo_major.txt" DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_wind_force test/test_force.cc)
  target_compile_options(test_wind_force PRIVATE -UNDEBUG)
  add_test(NAME wind_force COMMAND test_wind_force)
endif()
ament_package()
